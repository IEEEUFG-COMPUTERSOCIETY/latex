\newcommand{\CardFront}{%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[IEEEDarkBlue]
      (current page.south west) rectangle (current page.north east);
    \CardIcosahedrons
    \fill[IEEECSOrange]
      (current page.south west)
      rectangle ([yshift=2mm]current page.south east);
    \node[anchor=north west,inner sep=0pt]
      at ([xshift=5mm,yshift=-4mm]current page.north west) {%
        \IEEECSLogo[
          variant=orange-white,
          width=32mm,
          clear-space=false
        ]%
      };
    \node[anchor=north east,inner sep=0pt]
      at ([xshift=-5mm,yshift=-4mm]current page.north east) {%
        \UFGLogo[
          orientation=horizontal,
          variant=white,
          width=25mm,
          clear-space=false
        ]%
      };
    \node[
      anchor=south west,
      align=left,
      inner sep=0pt,
      text=IEEEWhite,
      text width=64mm
    ] at ([xshift=5mm,yshift=7mm]current page.south west) {%
      {\IEEECSHeadingFont\bfseries\fontsize{14}{15}\selectfont
        \CardName\par}
      \vspace{1.2mm}
      {\IEEECSBodyFont\fontsize{7.5}{9}\selectfont
        \CardRole\par}
    };
  \end{tikzpicture}%
  \null
}
